// Вызывается из регламентного задания по расписанию.
Процедура ВыполнитьЗагрузкуЗаказовССайта() Экспорт

	ПутьКаталога = СокрЛП(Константы.ПутьВходящихЗаказовИнтернетМагазина.Получить());
	Если ПустаяСтрока(ПутьКаталога) Тогда
		Возврат;
	КонецЕсли;

	КаталогНаДиске = Новый Файл(ПутьКаталога);
	Если НЕ КаталогНаДиске.Существует() Или НЕ КаталогНаДиске.ЭтоКаталог() Тогда
		ЗаписьЖурналаРегистрации(
			"ИнтернетМагазин.ЗагрузкаЗаказов",
			УровеньЖурналаРегистрации.Предупреждение,
			,,
			"Каталог входящих заказов не найден или не каталог: " + ПутьКаталога);
		Возврат;
	КонецЕсли;

	МассивФайлов = НайтиФайлы(ПутьКаталога, "order_*.json", Ложь);
	Если МассивФайлов.Количество() = 0 Тогда
		Возврат;
	КонецЕсли;

	КаталогОбработано = ОбъединитьПути(ПутьКаталога, "обработано");
	КаталогОбработаноФайл = Новый Файл(КаталогОбработано);
	Если НЕ КаталогОбработаноФайл.Существует() Тогда
		СоздатьКаталог(КаталогОбработано);
	КонецЕсли;

	Для Каждого ФайлЗаказа Из МассивФайлов Цикл

		ПолноеИмя = ФайлЗаказа.ПолноеИмя;

		Попытка

			ТекстДок = Новый ТекстовыйДокумент;
			ТекстДок.Прочитать(ПолноеИмя, КодировкаТекста.UTF8);
			СтрокаJSON = ТекстДок.ПолучитьТекст();

			Если ПустаяСтрока(СтрокаJSON) Тогда
				Продолжить;
			КонецЕсли;

			ЧтениеJSON = Новый ЧтениеJSON;
			ЧтениеJSON.УстановитьСтроку(СтрокаJSON);
			Данные = ПрочитатьJSON(ЧтениеJSON);
			ЧтениеJSON.Закрыть();

			ИдЗаказаСайта = "";
			Попытка
				ИдЗаказаСайта = Формат(Данные.id_заказа, "ЧГ=");
			Исключение
				ИдЗаказаСайта = "?";
			КонецПопытки;

			ЗаписьЖурналаРегистрации(
				"ИнтернетМагазин.ЗагрузкаЗаказов",
				УровеньЖурналаРегистрации.Информация,
				,,
				"Принят заказ с сайта № " + ИдЗаказаСайта + " (файл: " + ФайлЗаказа.Имя + ")");

			// -----------------------------------------------------------------
			// Здесь создайте и запишите документ вашей конфигурации по структуре Данные
			// (см. HOW_TO_IMPORT_IN_1C.txt). Пример каркаса закомментирован:
			//
			// Док = Документы.<ИмяВашегоДокумента>.СоздатьДокумент();
			// Док.Дата = ТекущаяДатаСеанса();
			// Док.Комментарий = "Сайт №" + ИдЗаказаСайта;
			// // ... контрагент из Данные.клиент, табличная часть из Данные.позиции ...
			// Док.Записать(РежимЗаписиДокумента.Запись);
			// -----------------------------------------------------------------

			ИмяВАрхиве = ОбъединитьПути(КаталогОбработано, ФайлЗаказа.Имя);
			ПереместитьФайл(ПолноеИмя, ИмяВАрхиве);

		Исключение

			ЗаписьЖурналаРегистрации(
				"ИнтернетМагазин.ЗагрузкаЗаказов",
				УровеньЖурналаРегистрации.Ошибка,
				,,
				"Ошибка обработки " + ПолноеИмя + ": " + ОписаниеОшибки());

	КонецЦикла;

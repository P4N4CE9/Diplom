Функция orderPOST(Запрос)

    Ответ.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");

    // Проверка секрета (как на сайте ONEC_WEBHOOK_SECRET)
    ОжидаемыйСекрет = ""; // лучше взять из константы конфигурации
    ЗаголовокАвторизации = Запрос.Заголовки.Получить("Authorization");
    Если ЗначениеЗаполнено(ОжидаемыйСекрет) Тогда
        Если ЗаголовокАвторизации <> "Bearer " + ОжидаемыйСекрет Тогда
            Ответ.УстановитьТелоИзСтроки("{""ok"":false,""error"":""unauthorized""}");
            Ответ.КодСостояния = 401;
            Возврат Ответ;
        КонецЕсли;
    КонецЕсли;

    Тело = Запрос.ПолучитьТелоКакСтроку();

        ЧтениеJSON = Новый ЧтениеJSON;
        ЧтениеJSON.УстановитьСтроку(Тело);
        Данные = ПрочитатьJSON(ЧтениеJSON);
    Исключение
        Ответ.УстановитьТелоИзСтроки("{""ok"":false,""error"":""invalid json""}");
        Ответ.КодСостояния = 400;
        Возврат Ответ;
    КонецПопытки;

    // Данные: id_заказа, дата_заказа, статус, способ_оплаты, общая_сумма,
    // клиент (id_клиента, фио, email, телефон, адрес), позиции[] (id_запчасти, артикул, название, количество, цена, сумма)

    НачатьТранзакцию();
    Попытка
        // --- Создать документ (ПРИМЕР: подставьте свой тип и реквизиты) ---
        // Док = Документы.ЗаказПокупателя.СоздатьДокумент();
        // Док.Дата = ТекущаяДатаСеанса();
        // Док.Комментарий = "Сайт id=" + Формат(Данные.id_заказа, "ЧГ=");
        // ... заполнение контрагента по Данные.клиент ...
        // ... заполнение табличной части из Данные.позиции ...
        // Док.Записать(РежимЗаписиДокумента.Проведение);
        // --- конец примера ---

        ЗафиксироватьТранзакцию();
    Исключение
        ОтменитьТранзакцию();

    КонецПопытки;
